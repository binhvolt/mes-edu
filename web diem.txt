  <script type="module">
    import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-app.js";
    import { getAuth, signInAnonymously } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-auth.js";
    import { getFirestore, doc, setDoc, onSnapshot } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-firestore.js";

    const firebaseConfig = {
      apiKey: "AIzaSyDkHugCuX1fS3H426CuGQJUa7z0ItGuEv0",
      authDomain: "web-study-167d7.firebaseapp.com",
      projectId: "web-study-167d7",
      storageBucket: "web-study-167d7.firebasestorage.app",
      messagingSenderId: "1090482112129",
      appId: "1:1090482112129:web:1f619da8687e88d3cfaf80",
      measurementId: "G-LC17YMTPMW"
    };

    const app = initializeApp(firebaseConfig);
    const auth = getAuth(app);
    const db = getFirestore(app);
    const stateDocRef = doc(db, "mes_edu_app", "main_state");

    let appData = { pin: "2026", students: [], tasks: [] };
    let isTeacherUnlocked = false;

    function setSyncStatus(text, colorClass) {
      const el = document.getElementById("syncStatus");
      el.textContent = text;
      el.className = `text-xs px-3 py-1.5 rounded-full text-white font-medium ${colorClass}`;
    }

    async function initFirebase() {
      try {
        await signInAnonymously(auth);
        onSnapshot(stateDocRef, (snap) => {
          if (snap.exists()) {
            appData = snap.data();
            if (!appData.students) appData.students = [];
            if (!appData.tasks) appData.tasks = [];
            if (!appData.pin) appData.pin = "2026";
          } else {
            setDoc(stateDocRef, appData);
          }
          refreshAllUI();
          setSyncStatus("Đã đồng bộ trực tuyến", "bg-emerald-600");
        }, (err) => {
          console.error(err);
          setSyncStatus("Lỗi quyền Firestore", "bg-rose-600");
        });
      } catch (err) {
        console.error(err);
        setSyncStatus("Sai cấu hình Firebase", "bg-rose-600");
      }
    }

    async function saveData() {
      setSyncStatus("Đang lưu...", "bg-amber-600");
      try {
        await setDoc(stateDocRef, appData);
        setSyncStatus("Đã đồng bộ trực tuyến", "bg-emerald-600");
      } catch (e) {
        setSyncStatus("Lỗi lưu dữ liệu", "bg-rose-600");
        alert("Không lưu được dữ liệu. Kiểm tra lại Firestore Rules hoặc mạng!");
      }
    }

    window.switchRole = (role) => {
      document.getElementById("studentView").classList.toggle("hidden", role !== "student");
      document.getElementById("teacherView").classList.toggle("hidden", role !== "teacher");
      document.getElementById("btnRoleStudent").className = role === "student"
        ? "px-3.5 py-1.5 rounded-lg text-sm font-medium bg-white text-indigo-700 shadow-sm"
        : "px-3.5 py-1.5 rounded-lg text-sm font-medium bg-indigo-600 text-white";
      document.getElementById("btnRoleTeacher").className = role === "teacher"
        ? "px-3.5 py-1.5 rounded-lg text-sm font-medium bg-white text-indigo-700 shadow-sm"
        : "px-3.5 py-1.5 rounded-lg text-sm font-medium bg-indigo-600 text-white";
    };

    window.openTeacherAuth = () => {
      if (isTeacherUnlocked) {
        window.switchRole("teacher");
      } else {
        document.getElementById("pinInput").value = "";
        document.getElementById("pinModal").classList.remove("hidden");
        document.getElementById("pinInput").focus();
      }
    };
    window.closePinModal = () => document.getElementById("pinModal").classList.add("hidden");
    window.verifyPin = () => {
      const val = document.getElementById("pinInput").value.trim();
      if (val === appData.pin) {
        isTeacherUnlocked = true;
        window.closePinModal();
        window.switchRole("teacher");
      } else {
        alert("Sai mã PIN!");
      }
    };

    window.switchTeacherTab = (tab) => {
      ["tasks", "students", "report", "settings"].forEach(t => {
        document.getElementById("panel" + t.charAt(0).toUpperCase() + t.slice(1)).classList.toggle("hidden", t !== tab);
        document.getElementById("tab" + t.charAt(0).toUpperCase() + t.slice(1)).className = t === tab
          ? "px-4 py-2 rounded-xl text-sm font-semibold bg-indigo-600 text-white"
          : "px-4 py-2 rounded-xl text-sm font-semibold text-slate-600 hover:bg-slate-100";
      });
      if (tab === "report") window.generateZaloReport();
    };

    function refreshAllUI() {
      const currentStudent = document.getElementById("studentSelect").value;
      const optionsHtml = `<option value="">-- Chọn tên học sinh --</option>` +
        appData.students.map(s => `<option value="${s.id}">${s.name} (${s.grade})</option>`).join("");

      document.getElementById("studentSelect").innerHTML = optionsHtml;
      document.getElementById("studentSelect").value = currentStudent;

      document.getElementById("taskStudentId").innerHTML =
        appData.students.map(s => `<option value="${s.id}">${s.name} (${s.grade})</option>`).join("");

      document.getElementById("filterTeacherStudent").innerHTML =
        `<option value="ALL">Tất cả học sinh</option>` +
        appData.students.map(s => `<option value="${s.id}">${s.name}</option>`).join("");

      document.getElementById("reportStudentSelect").innerHTML =
        appData.students.map(s => `<option value="${s.id}">${s.name} (${s.grade})</option>`).join("");

      renderStudentManager();
      window.renderTeacherTasks();
      window.renderStudentTasks();
    }

    window.addStudent = async () => {
      const name = document.getElementById("newStudentName").value.trim();
      const grade = document.getElementById("newStudentGrade").value.trim() || "Chưa rõ lớp";
      if (!name) return alert("Vui lòng nhập tên học sinh!");
      appData.students.push({ id: "st_" + Date.now(), name, grade });
      document.getElementById("newStudentName").value = "";
      document.getElementById("newStudentGrade").value = "";
      await saveData();
    };

    window.deleteStudent = async (id) => {
      if (!confirm("Xóa học sinh này và toàn bộ bài tập liên quan?")) return;
      appData.students = appData.students.filter(s => s.id !== id);
      appData.tasks = appData.tasks.filter(t => t.studentId !== id);
      await saveData();
    };

    function renderStudentManager() {
      const container = document.getElementById("studentManagerList");
      if (appData.students.length === 0) {
        container.innerHTML = `<p class="text-sm text-slate-400 py-4">Chưa có học sinh nào.</p>`;
        return;
      }
      container.innerHTML = appData.students.map(s => `
        <div class="py-3 flex items-center justify-between">
          <div>
            <p class="font-semibold text-sm text-slate-800">${s.name}</p>
            <p class="text-xs text-slate-500">${s.grade}</p>
          </div>
          <button onclick="deleteStudent('${s.id}')" class="text-xs text-rose-600 hover:underline font-medium">Xóa</button>
        </div>
      `).join("");
    }

    window.addTask = async () => {
      const studentId = document.getElementById("taskStudentId").value;
      const subject = document.getElementById("taskSubject").value;
      const title = document.getElementById("taskTitle").value.trim();
      const deadline = document.getElementById("taskDeadline").value;
      const note = document.getElementById("taskNote").value.trim();

      if (!studentId) return alert("Hãy thêm học sinh trước!");
      if (!title || !deadline) return alert("Vui lòng nhập tên bài tập và hạn chót!");

      appData.tasks.unshift({
        id: "tk_" + Date.now(),
        studentId,
        subject,
        title,
        deadline,
        note,
        status: "PENDING",
        submitLink: "",
        studentNote: "",
        submittedAt: "",
        score: "",
        feedback: ""
      });

      document.getElementById("taskTitle").value = "";
      document.getElementById("taskNote").value = "";
      await saveData();
    };

    window.deleteTask = async (id) => {
      if (!confirm("Xóa bài tập này?")) return;
      appData.tasks = appData.tasks.filter(t => t.id !== id);
      await saveData();
    };

    function formatDeadline(dtStr) {
      if (!dtStr) return "";
      const d = new Date(dtStr);
      return d.toLocaleString("vi-VN", { hour: "2-digit", minute: "2-digit", day: "2-digit", month: "2-digit", year: "numeric" });
    }

    function isOverdue(task) {
      if (task.status !== "PENDING") return false;
      return new Date(task.deadline).getTime() < Date.now();
    }

    window.renderStudentTasks = () => {
      const stId = document.getElementById("studentSelect").value;
      const statsEl = document.getElementById("studentStats");
      const listEl = document.getElementById("studentTaskList");

      if (!stId) {
        statsEl.classList.add("hidden");
        listEl.innerHTML = `<div class="text-center py-12 text-slate-400 bg-white rounded-2xl border border-dashed border-slate-300">Vui lòng chọn tên học sinh ở phía trên để hiển thị danh sách deadline.</div>`;
        return;
      }

      const myTasks = appData.tasks.filter(t => t.studentId === stId);
      statsEl.classList.remove("hidden");

      const overdueCount = myTasks.filter(t => isOverdue(t)).length;
      const pendingCount = myTasks.filter(t => t.status === "PENDING").length;
      const doneCount = myTasks.filter(t => t.status !== "PENDING").length;

      document.getElementById("stTotal").textContent = myTasks.length;
      document.getElementById("stPending").textContent = pendingCount;
      document.getElementById("stOverdue").textContent = overdueCount;
      document.getElementById("stDone").textContent = doneCount;

      if (myTasks.length === 0) {
        listEl.innerHTML = `<div class="text-center py-10 text-slate-400 bg-white rounded-2xl border border-slate-200">Hiện tại em chưa có bài tập nào được giao.</div>`;
        return;
      }

      listEl.innerHTML = myTasks.map(t => {
        const overdue = isOverdue(t);
        let badge = `<span class="px-2.5 py-1 rounded-full text-xs font-semibold bg-amber-100 text-amber-700">Chưa nộp</span>`;
        if (overdue) badge = `<span class="px-2.5 py-1 rounded-full text-xs font-semibold bg-rose-100 text-rose-700">Quá hạn</span>`;
        if (t.status === "SUBMITTED") badge = `<span class="px-2.5 py-1 rounded-full text-xs font-semibold bg-sky-100 text-sky-700">Đã nộp - Chờ chấm</span>`;
        if (t.status === "GRADED") badge = `<span class="px-2.5 py-1 rounded-full text-xs font-semibold bg-emerald-100 text-emerald-700">Đã chấm: ${t.score}</span>`;

        return `
          <div class="bg-white p-5 rounded-2xl shadow-sm border ${overdue ? 'border-rose-300' : 'border-slate-200'} flex flex-col sm:flex-row sm:items-center justify-between gap-4">
            <div class="space-y-1.5">
              <div class="flex flex-wrap items-center gap-2">
                <span class="px-2.5 py-0.5 rounded-md text-xs font-bold bg-indigo-50 text-indigo-700">${t.subject}</span>
                ${badge}
                <span class="text-xs text-slate-500">Hạn chót: <b>${formatDeadline(t.deadline)}</b></span>
              </div>
              <h4 class="font-bold text-base text-slate-800">${t.title}</h4>
              ${t.note ? `<p class="text-xs text-slate-600 bg-slate-50 p-2 rounded-lg border border-slate-100">${t.note}</p>` : ""}
              ${t.submitLink ? `<p class="text-xs text-sky-700">Bài đã nộp: <a href="${t.submitLink}" target="_blank" class="underline font-medium">Mở link bài làm</a></p>` : ""}
              ${t.feedback ? `<p class="text-xs text-emerald-800 bg-emerald-50 p-2 rounded-lg border border-emerald-200"><b>Nhận xét:</b> ${t.feedback}</p>` : ""}
            </div>
            <div class="shrink-0">
              <button onclick="openSubmitModal('${t.id}')"
                class="px-4 py-2 rounded-xl text-sm font-semibold ${t.status === 'PENDING' ? 'bg-indigo-600 hover:bg-indigo-700 text-white' : 'bg-slate-100 hover:bg-slate-200 text-slate-700'}">
                ${t.status === 'PENDING' ? 'Nộp Bài' : 'Nộp Lại'}
              </button>
            </div>
          </div>
        `;
      }).join("");
    };

    window.openSubmitModal = (taskId) => {
      const task = appData.tasks.find(t => t.id === taskId);
      document.getElementById("submitTaskId").value = taskId;
      document.getElementById("submitLink").value = task?.submitLink || "";
      document.getElementById("submitStudentNote").value = task?.studentNote || "";
      document.getElementById("submitModal").classList.remove("hidden");
    };
    window.closeSubmitModal = () => document.getElementById("submitModal").classList.add("hidden");

    window.confirmSubmitTask = async () => {
      const id = document.getElementById("submitTaskId").value;
      const link = document.getElementById("submitLink").value.trim();
      const studentNote = document.getElementById("submitStudentNote").value.trim();
      if (!link) return alert("Vui lòng dán link bài làm (Google Drive / Ảnh)!");

      const idx = appData.tasks.findIndex(t => t.id === id);
      if (idx !== -1) {
        appData.tasks[idx].status = "SUBMITTED";
        appData.tasks[idx].submitLink = link;
        appData.tasks[idx].studentNote = studentNote;
        appData.tasks[idx].submittedAt = new Date().toLocaleString("vi-VN");
      }
      window.closeSubmitModal();
      await saveData();
    };

    window.renderTeacherTasks = () => {
      const filterSt = document.getElementById("filterTeacherStudent").value;
      const filterStatus = document.getElementById("filterTeacherStatus").value;
      const listEl = document.getElementById("teacherTaskList");

      let filtered = appData.tasks;
      if (filterSt !== "ALL") filtered = filtered.filter(t => t.studentId === filterSt);
      if (filterStatus !== "ALL") filtered = filtered.filter(t => t.status === filterStatus);

      if (filtered.length === 0) {
        listEl.innerHTML = `<div class="bg-white p-8 rounded-2xl border border-slate-200 text-center text-sm text-slate-400">Không có bài tập nào phù hợp bộ lọc.</div>`;
        return;
      }

      listEl.innerHTML = filtered.map(t => {
        const st = appData.students.find(s => s.id === t.studentId);
        const overdue = isOverdue(t);
        let statusTag = `<span class="px-2 py-0.5 rounded text-xs font-semibold bg-amber-100 text-amber-700">Chưa nộp</span>`;
        if (overdue) statusTag = `<span class="px-2 py-0.5 rounded text-xs font-semibold bg-rose-100 text-rose-700">Quá hạn</span>`;
        if (t.status === "SUBMITTED") statusTag = `<span class="px-2 py-0.5 rounded text-xs font-semibold bg-sky-100 text-sky-700">Đã nộp (${t.submittedAt})</span>`;
        if (t.status === "GRADED") statusTag = `<span class="px-2 py-0.5 rounded text-xs font-semibold bg-emerald-100 text-emerald-700">Đã chấm: ${t.score}</span>`;

        return `
          <div class="bg-white p-4 rounded-2xl shadow-sm border border-slate-200 space-y-2">
            <div class="flex flex-wrap items-center justify-between gap-2">
              <div class="flex items-center gap-2">
                <span class="font-bold text-sm text-indigo-700">${st ? st.name : "Học sinh đã xóa"}</span>
                <span class="text-xs px-2 py-0.5 bg-slate-100 rounded font-medium">${t.subject}</span>
                ${statusTag}
              </div>
              <span class="text-xs text-slate-500">Hạn: ${formatDeadline(t.deadline)}</span>
            </div>
            <p class="font-semibold text-sm text-slate-800">${t.title}</p>
            ${t.submitLink ? `<p class="text-xs text-sky-700">Link bài nộp: <a href="${t.submitLink}" target="_blank" class="underline font-semibold">Xem bài của học sinh</a> ${t.studentNote ? `— <i>"${t.studentNote}"</i>` : ""}</p>` : ""}
            ${t.feedback ? `<p class="text-xs text-emerald-700">Lời phê: ${t.feedback}</p>` : ""}
            <div class="pt-2 flex justify-end gap-2 border-t border-slate-100">
              <button onclick="openGradeModal('${t.id}')" class="px-3 py-1.5 rounded-lg text-xs font-semibold bg-indigo-50 text-indigo-700 hover:bg-indigo-100">
                Chấm điểm / Nhận xét
              </button>
              <button onclick="deleteTask('${t.id}')" class="px-3 py-1.5 rounded-lg text-xs font-semibold bg-rose-50 text-rose-600 hover:bg-rose-100">
                Xóa
              </button>
            </div>
          </div>
        `;
      }).join("");
    };

    window.openGradeModal = (taskId) => {
      const task = appData.tasks.find(t => t.id === taskId);
      document.getElementById("gradeTaskId").value = taskId;
      document.getElementById("gradeScore").value = task?.score || "";
      document.getElementById("gradeFeedback").value = task?.feedback || "";
      document.getElementById("gradeModal").classList.remove("hidden");
    };
    window.closeGradeModal = () => document.getElementById("gradeModal").classList.add("hidden");

    window.confirmGradeTask = async () => {
      const id = document.getElementById("gradeTaskId").value;
      const score = document.getElementById("gradeScore").value.trim();
      const feedback = document.getElementById("gradeFeedback").value.trim();
      if (!score) return alert("Vui lòng nhập điểm hoặc đánh giá!");

      const idx = appData.tasks.findIndex(t => t.id === id);
      if (idx !== -1) {
        appData.tasks[idx].status = "GRADED";
        appData.tasks[idx].score = score;
        appData.tasks[idx].feedback = feedback;
      }
      window.closeGradeModal();
      await saveData();
    };

    window.generateZaloReport = () => {
      const stId = document.getElementById("reportStudentSelect").value;
      const st = appData.students.find(s => s.id === stId);
      if (!st) {
        document.getElementById("zaloReportText").value = "Chưa có học sinh để tạo báo cáo.";
        return;
      }
      const tasks = appData.tasks.filter(t => t.studentId === stId);
      let lines = [
        `[MES EDU - BÁO CÁO TIẾN ĐỘ HỌC TẬP]`,
        `Học sinh: ${st.name} (${st.grade})`,
        `Người phụ trách: Nguyễn Văn Việt Bình`,
        `-----------------------------------`
      ];
      if (tasks.length === 0) {
        lines.push("Hiện chưa có bài tập nào được ghi nhận.");
      } else {
        tasks.forEach((t, i) => {
          let stText = "Chưa nộp";
          if (isOverdue(t)) stText = "QUÁ HẠN (Chưa nộp)";
          if (t.status === "SUBMITTED") stText = "Đã nộp (Chờ chấm)";
          if (t.status === "GRADED") stText = `Đã hoàn thành — Kết quả: ${t.score}`;
          lines.push(`${i + 1}. [${t.subject}] ${t.title}`);
          lines.push(`   - Hạn chót: ${formatDeadline(t.deadline)}`);
          lines.push(`   - Trạng thái: ${stText}`);
          if (t.feedback) lines.push(`   - Nhận xét: ${t.feedback}`);
        });
      }
      document.getElementById("zaloReportText").value = lines.join("\n");
    };

    window.copyZaloReport = () => {
      const el = document.getElementById("zaloReportText");
      el.select();
      document.execCommand("copy");
      alert("Đã copy nội dung báo cáo! Bạn có thể dán thẳng vào Zalo.");
    };

      appData.pin = newPin;
      document.getElementById("newPinInput").value = "";
      await saveData();
      alert("Đã đổi mã PIN thành công!");
    };
